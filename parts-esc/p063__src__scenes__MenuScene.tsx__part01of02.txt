/** Shared sound reactions for interactive cats (menu + lab). */
export function catSound(e: CatEvent) {
  switch (e) {
    case "poke":
      sfx.meow();
      break;
    case "pokeBelly":
      sfx.giggle();
      break;
    case "pokeTail":
      sfx.grumble();
      break;
    case "pet":
      sfx.purr();
      break;
  }
}
