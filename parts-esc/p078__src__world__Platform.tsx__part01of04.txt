const boxCache = new Map\u003cstring, THREE.BufferGeometry\u003e();
export function box(w: number, h: number, d: number, r: number) {
  const key = `${w.toFixed(2)}|${h}|${d}|${r}`;
  let g = boxCache.get(key);
  if (!g) {
    g = new RoundedBoxGeometry(w, h, d, 5, r);
    boxCache.set(key, g);
  }
  return g;
}
